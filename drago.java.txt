public class Drago implements Runnable {
    private int battiti;

    public Drago(int battiti) {
        this.battiti = battiti;
    }

    @Override
    public void run() {
        try {
            for (int i = 1; i <= battiti; i++) {
                System.out.println("Drago: battito d'ali numero " + i);
                Thread.sleep(1000); 
            }
        } catch (InterruptedException e) {
            System.out.println(">>> Il drago è stato colpito dal cacciatore e muore definitivamente! <<<");
        }
    }
}
