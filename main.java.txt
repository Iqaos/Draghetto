import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        
        System.out.print("Inserisci il numero di battiti d'ali del drago: ");
        int battiti = scanner.nextInt();
        scanner.close();

        Drago dragoTask = new Drago(battiti);
        Thread threadDrago = new Thread(dragoTask);
        
        threadDrago.start();

        try {
            for (int x = 1; x <= battiti; x++) {
                System.out.println("Main: sto osservando il drago... (controllo " + x + ")");
                Thread.sleep(400);
            }
        } catch (InterruptedException e) {
            System.out.println("Il main è stato interrotto.");
        }

        Cacciatore cacciatore = new Cacciatore();
        cacciatore.colpisciDrago(threadDrago);
        
        try {
            threadDrago.join();
        } catch (InterruptedException e) {
            e.printStackTrace();
        }
    }
}
